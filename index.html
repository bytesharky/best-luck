/**
  人生就像一段程序，沿着主线稳定推进，
  直到那次不愿 rollback 的 exception，
  惟愿选择 keep，往后余生再无 break。
**/
#include <lifetime> // 一生
#include <limits>   // 极限
#include <optional> // 可选 (你曾不是必然，如今是全部)

struct Life {
    // 在遇见你之前，我不知道什么是最好的
    Best best = Best::undefined(); 
    // 运气值为零
    Luck luck = 0;
    // 但心还活着
    bool heartbeat = true;
    bool alive() const { return heartbeat; }
};

int main() {
    Life me;

    try {
        // 原本的我平淡、按部就班的生活
        me.runAsPlanned();
    }
    // 你的出现打乱了一切
    catch (const You& you) {
        // 独一无二的你，胜过我以往所有的"最好"
        me.best = std::max(me.best, you);
        // 惟愿 keep （把你留在身边），不愿 rollback (回到从前)
        me.keep(you);
    }
    // 遇见你本是极低概率事件(难得的缘分)
    if (meet(you)) [[unlikely]] {
        // 如果遇见你，是我今生最大辛运
        me.luck = std::numeric_limits<Luck>::max();
    }

    // 有生之年
    while (me.alive()) {
        // 一直珍惜你，再无 break (不离不弃)
        me.cherish(you);
    }

    // 直到一生圆满、无遗憾
    return 0;
}
